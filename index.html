let lastSpin = localStorage.getItem(user + '_last_spin'); if(lastSpin) { let diff = new Date().getTime() - parseInt(lastSpin); let dayMs = 24 * 60 * 60 * 1000; if(diff < dayMs) { alert("عذراً، يمكنك تدوير العجلة مرة واحدة فقط كل 24 ساعة!"); checkTimer(); return; } } isSpinning = true; const wheel = document.getElementById('wheel'); let prizes = getPrizes(); // تصفية الجوائز واستبعاد أي جائزة وزنها صفر أو أقل تماماً من العشوائيات لتجنب الخسارة let validPrizes = prizes.filter(p => (p.weight || 0) > 0); if(validPrizes.length === 0) { alert("عجلة الحظ لا تحتوي على جوائز متاحة للتدوير حالياً!"); isSpinning = false; return; } let totalWeight = validPrizes.reduce((sum, p) => sum + p.weight, 0); let randomWeight = Math.random() * totalWeight; let currentSum = 0; let winningPrize = validPrizes[0]; for(let i = 0; i < validPrizes.length; i++) { currentSum += validPrizes[i].weight; if(randomWeight <= currentSum) { winningPrize = validPrizes[i]; break; } } let winningIndex = prizes.findIndex(p => p.name === winningPrize.name); let angleStep = 360 / prizes.length; let targetAngle = 360 - (winningIndex * angleStep) - (angleStep / 2); wheel.style.transform = rotate(${360 * 8 + targetAngle}deg); setTimeout(() => { isSpinning = false; localStorage.setItem(user + '_last_spin', new Date().getTime().toString()); let wonPrizeName = ${winningPrize.icon || '🎁'} ${winningPrize.name}; let myPrizes = JSON.parse(localStorage.getItem(user + '_my_prizes') || '[]'); myPrizes.push({ name: wonPrizeName, status: "فوز مباشر من العجلة ✅", contact: "-" }); localStorage.setItem(user + '_my_prizes', JSON.stringify(myPrizes)); let logs = JSON.parse(localStorage.getItem(user + '_logs') || '[]'); logs.unshift(🎡 ربحت من العجلة: (${wonPrizeName})); localStorage.setItem(user + '_logs', JSON.stringify(logs)); checkUserState(); showToast(تهانينا! فزت بـ: (${wonPrizeName}) 🎉); checkTimer(); }, 5000); }
